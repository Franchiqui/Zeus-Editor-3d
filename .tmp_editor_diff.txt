+/**
+ * Arco de redondeo de la esquina `i` calculado sobre la forma ACTUAL del
+ * polígono. Si una arista vecina ya está curvada (el vecino tiene asas Bézier),
+ * el arco sale TANGENTE a esa curva en vez de a la recta entre vértices, de
+ * modo que el redondeo empalma con las curvas existentes y no aparece un codo
+ * ni una "hinchazón" en la unión. Con vecinos de esquina recta el resultado es
+ * idéntico al redondeo clásico.
+ */
+function cornerArcPoints(
+  poly: Polygon,
+  i: number,
+  curveAmount: number,
+  segments: number
+): Point2D[] | null {
+  const n = poly.length;
+  if (n < 3) return null;
+  const prev = poly[(i - 1 + n) % n];
+  const vert = poly[i];
+  const next = poly[(i + 1) % n];
+  const radius = polygonRoundRadius(poly, curveAmount);
+  // Sin curvas alrededor: redondeo recto de siempre.
+  if (!prev.hOut && !next.hIn) {
+    return computeRoundCornerPoints(prev, vert, next, radius, segments);
+  }
+  // Punto de control REAL de cada arista en el extremo del vértice: si el
+  // vecino tiene asa, marca la tangente de su curva; si no, la recta.
+  const prevRef = prev.hOut ?? prev;
+  const nextRef = next.hIn ?? next;
+  let u1x = prevRef.x - vert.x;
+  let u1y = prevRef.y - vert.y;
+  let u2x = nextRef.x - vert.x;
+  let u2y = nextRef.y - vert.y;
+  const l1 = Math.hypot(u1x, u1y);
+  const l2 = Math.hypot(u2x, u2y);
+  if (l1 < 1e-6 || l2 < 1e-6) return null;
+  u1x /= l1;
+  u1y /= l1;
+  u2x /= l2;
+  u2y /= l2;
+  const cross = u1x * u2y - u1y * u2x;
+  if (Math.abs(cross) < 1e-3) return null;
+  const straight1 = Math.hypot(vert.x - prev.x, vert.y - prev.y);
+  const straight2 = Math.hypot(next.x - vert.x, next.y - vert.y);
+  const r = Math.min(radius, straight1 * 0.5, straight2 * 0.5);
+  if (r < 0.005) return null;
+  const t1 = { x: vert.x + u1x * r, y: vert.y + u1y * r };
+  const t2 = { x: vert.x + u2x * r, y: vert.y + u2y * r };
+  const dx = t2.x - t1.x;
+  const dy = t2.y - t1.y;
+  const s = (dx * u2y - dy * u2x) / cross;
+  const cx = t1.x + u1x * s;
+  const cy = t1.y + u1y * s;
+  const segCount = Math.max(1, Math.round(segments));
+  const arc: Point2D[] = [t1];
+  for (let k = 1; k <= segCount; k++) {
+    const t = k / (segCount + 1);
+    const it = 1 - t;
+    arc.push({
+      x: it * it * t1.x + 2 * it * t * cx + t * t * t2.x,
+      y: it * it * t1.y + 2 * it * t * cy + t * t * t2.y,
+    });
+  }
+  arc.push(t2);
+  return arc;
+}

+   * escribir la Y a mano en la vista ampliada.
+   */
+  const moveMeshSectionY = useCallback(
+    (id: number, delta: number) => {
+      setMeshSections((prev) =>
+        prev.map((s) => {
+          if (s.id !== id) return s;
+          const y = Math.min(1, Math.max(0, Number((s.y + delta).toFixed(2))));
+          return {
+            ...s,
+            y,
+            polygon: fitSectionToViews(
+              s.polygon,
+              y,
+              appliedMeshSilhouette,
+              appliedMeshSideView
+            ),
+          };
+        })
+      );
+    },
+    [appliedMeshSilhouette, appliedMeshSideView]
+  );
+
+  /** Opacidad de una plantilla en la vista superpuesta (0..1). */
+  const overlayOpacityOf = useCallback(
+    (id: number) => meshOverlayOpacities[id] ?? 0.6,
+    [meshOverlayOpacities]
+  );
+
+  const setOverlayOpacity = useCallback((id: number, value: number) => {
+    setMeshOverlayOpacities((prev) => ({ ...prev, [id]: value }));
+  }, []);
+
+  const resetOverlayOpacities = useCallback(() => {
+    setMeshOverlayOpacities({});
+  }, []);
+
+  /**
+   * Pasa coordenadas de pantalla a coordenadas del viewBox 0..100. El SVG usa
+   * viewBox "0 0 100 100" con preserveAspectRatio "meet", así que hay que
+   * descontar el letterbox para acertar con el punto pulsado. NO tiene en
+   * cuenta el zoom ni el encuadre: es la base de todo lo demás.
+   */
+  const overlayClientToViewBox = useCallback(
+    (clientX: number, clientY: number) => {
+      const el =
+        typeof document !== 'undefined'
+          ? (document.elementFromPoint(clientX, clientY) as HTMLElement | null)
+          : null;
+      const box = el?.closest('[data-overlay-canvas]') as HTMLElement | null;
+      if (!box) return { x: 50, y: 50 };
+      const r = box.getBoundingClientRect();
+      const scale = Math.min(r.width, r.height) / 100 || 1;
+      const offX = (r.width - 100 * scale) / 2;
+      const offY = (r.height - 100 * scale) / 2;
+      return {
+        x: (clientX - r.left - offX) / scale,
+        y: (clientY - r.top - offY) / scale,
+      };
+    },
+    []
+  );
+
+  /** Zoom alrededor de un punto del viewBox (deja ese punto fijo en pantalla). */
+  const overlayZoomAt = useCallback(
+    (factor: number, fv: { x: number; y: number }) => {
+      const z0 = meshOverlayZoom;
+      const z = Math.min(8, Math.max(0.5, z0 * factor));
+      if (z === z0) return;
+      const c = 50;
+      const p = meshOverlayPan;
+      setMeshOverlayZoom(z);
+      setMeshOverlayPan({
+        x: fv.x - c - (z / z0) * (fv.x - p.x - c),
+        y: fv.y - c - (z / z0) * (fv.y - p.y - c),
+      });
+    },
+    [meshOverlayZoom, meshOverlayPan]
+  );
+
+  /** Devuelve el lienzo a su encuadre original (zoom 100%, centrado). */
+  const overlayResetView = useCallback(() => {
+    setMeshOverlayZoom(1);
+    setMeshOverlayPan({ x: 0, y: 0 });
+  }, []);
+
+  // Refs siempre al día para el listener de la rueda, que se engancha una
+  // sola vez por lienzo (no pasivo) para frenar el scroll de la página.
+  const overlayClientToViewBoxRef = useRef(overlayClientToViewBox);
+  const overlayZoomAtRef = useRef(overlayZoomAt);
+  useEffect(() => {
+    overlayClientToViewBoxRef.current = overlayClientToViewBox;
+  }, [overlayClientToViewBox]);
+  useEffect(() => {
+    overlayZoomAtRef.current = overlayZoomAt;
+  }, [overlayZoomAt]);
+
+  /**
+   * Callback ref del contenedor del lienzo superpuesto: engancha la RUEDA como
+   * listener no pasivo para hacer zoom con el ratón sin que se desplace la
+   * página. Las coordenadas se resuelven con elementFromPoint + data-overlay-canvas.
+   */
+  const setOverlayCanvas = useCallback((el: HTMLDivElement | null) => {
+    if (!el) return;
+    const onWheel = (e: WheelEvent) => {
+      e.preventDefault();
+      const fv = overlayClientToViewBoxRef.current(e.clientX, e.clientY);
+      overlayZoomAtRef.current(e.deltaY < 0 ? 1.15 : 1 / 1.15, fv);
+    };
+    el.addEventListener('wheel', onWheel, { passive: false });
+  }, []);
+
+  /**
+   * Pasa coordenadas de pantalla a coordenadas de la MALLA 0..100, ya con el
+   * zoom y el encuadre deshechos (lo que usan el arrastre y el doble clic).
+   */
+  const overlayClientToView = useCallback(
+    (clientX: number, clientY: number) => {
+      const v = overlayClientToViewBox(clientX, clientY);
+      const c = 50;
+      const z = meshOverlayZoom || 1;
+      const p = meshOverlayPan;
+      return { x: (v.x - p.x - c) / z + c, y: (v.y - p.y - c) / z + c };
+    },
+    [overlayClientToViewBox, meshOverlayZoom, meshOverlayPan]
+  );
+
+  /** Empieza a desplazar el lienzo al arrastrar el fondo. */
+  const overlayBeginPan = useCallback(
+    (clientX: number, clientY: number) => {
+      const v = overlayClientToViewBox(clientX, clientY);
+      overlayPanRef.current = {
+        viewX: v.x,
+        viewY: v.y,
+        panX: meshOverlayPan.x,
+        panY: meshOverlayPan.y,
+      };
+    },
+    [overlayClientToViewBox, meshOverlayPan]
+  );
+
+  /** Aplica el imán (rejilla) a un valor del lienzo 0..100 si está activo. */
+  const overlaySnapView = useCallback(
+    (v: number) => {
+      if (!meshOverlaySnap || meshOverlaySnapStep <= 0) return v;
+      const s = meshOverlaySnapStep * 100;
+      return Math.round(v / s) * s;
+    },
+    [meshOverlaySnap, meshOverlaySnapStep]
+  );
+
+  /** Vértices de la malla que caen dentro del rectángulo (malla 0..100). */
+  const overlayApplyMarquee = useCallback(
+    (x0: number, y0: number, x1: number, y1: number) => {
+      const editId = meshOverlayEditId;
+      if (editId == null) return;
+      const section = meshSections.find((s) => s.id === editId);
+      if (!section) return;
+      const minX = Math.min(x0, x1);
+      const maxX = Math.max(x0, x1);
+      const minY = Math.min(y0, y1);
+      const maxY = Math.max(y0, y1);
+      const inside: number[] = [];
+      section.polygon.forEach((p, i) => {
+        const vx = p.x * 100;
+        const vy = p.y * 100;
+        if (vx >= minX && vx <= maxX && vy >= minY && vy <= maxY) inside.push(i);
+      });
+      setMeshOverlaySelection(inside);
+    },
+    [meshOverlayEditId, meshSections]
+  );
+
+  /** Empieza la marquesina de selección (herramienta Selección). */
+  const overlayBeginMarquee = useCallback(
+    (clientX: number, clientY: number) => {
+      const v = overlayClientToView(clientX, clientY);
+      overlayMarqueeRef.current = { x0: v.x, y0: v.y, x1: v.x, y1: v.y };
+      setMeshOverlayMarquee({ x0: v.x, y0: v.y, x1: v.x, y1: v.y });
+      setMeshOverlaySelection([]);
+    },
+    [overlayClientToView]
+  );
+
+  /** Mueve en grupo los vértices seleccionados según el puntero. */
+  const overlayMoveGroup = useCallback(
+    (clientX: number, clientY: number) => {
+      const drag = overlayGroupDragRef.current;
+      if (!drag) return;
+      const { x, y } = overlayClientToView(clientX, clientY);
+      let dx = x - drag.startView.x;
+      let dy = y - drag.startView.y;
+      if (meshOverlaySnap && meshOverlaySnapStep > 0) {
+        const s = meshOverlaySnapStep * 100;
+        dx = Math.round(dx / s) * s;
+        dy = Math.round(dy / s) * s;
+      }
+      const moved = new Set(drag.vertices);
+      const verts: Point2D[] = drag.startPoly.map((p, i) => {
+        if (!moved.has(i)) return { ...p };
+        const nx = Math.min(100, Math.max(0, p.x * 100 + dx)) / 100;
+        const ny = Math.min(100, Math.max(0, p.y * 100 + dy)) / 100;
+        const v: Point2D = { ...p, x: nx, y: ny };
+        const odx = nx - p.x;
+        const ody = ny - p.y;
+        if (p.hIn) v.hIn = { x: p.hIn.x + odx, y: p.hIn.y + ody };
+        if (p.hOut) v.hOut = { x: p.hOut.x + odx, y: p.hOut.y + ody };
+        return v;
+      });
+      setMeshSections((prev) =>
+        prev.map((s) => (s.id === drag.id ? { ...s, polygon: verts } : s))
+      );
+    },
+    [overlayClientToView, meshOverlaySnap, meshOverlaySnapStep]
+  );
+
+  /** Elimina el vértice indicado si aún quedan al menos 3. */
+  const overlayRemoveVertexAt = useCallback(
+    (vertexIndex: number) => {
+      const editId = meshOverlayEditId;
+      if (editId == null) return;
+      setMeshSections((prev) =>
+        prev.map((s) => {
+          if (s.id !== editId) return s;
+          if (s.polygon.length <= 3) return s;
+          return {
+            ...s,
+            polygon: s.polygon.filter((_, i) => i !== vertexIndex),
+          };
+        })
+      );
+      setMeshOverlaySelection((sel) =>
+        sel
+          .filter((i) => i !== vertexIndex)
+          .map((i) => (i > vertexIndex ? i - 1 : i))
+      );
+      setMeshOverlayHoverVertex(null);
+    },
+    [meshOverlayEditId]
+  );
+
+  /** Redondea en arco la esquina del vértice (sustituye el vértice por puntos). */
+  const overlayRoundCornerAt = useCallback(
+    (vertexIndex: number) => {
+      const editId = meshOverlayEditId;
+      if (editId == null) return;
+      setMeshSections((prev) =>
+        prev.map((s) => {
+          if (s.id !== editId) return s;
+          const poly = s.polygon;
+          const n = poly.length;
+          if (n < 3) return s;
+          const pts = cornerArcPoints(
+            poly,
+            vertexIndex,
+            meshOverlayCurveAmount,
+            meshOverlayCurveSegments
+          );
+          if (!pts || pts.length < 2) return s;
+          const verts = poly.map((p) => ({ ...p }));
+          verts.splice(
+            vertexIndex,
+            1,
+            ...pts.map((q) => ({ x: q.x, y: q.y }))
+          );
+          return { ...s, polygon: verts };
+        })
+      );
+      setMeshOverlaySelection([]);
+      setMeshOverlayHoverVertex(null);
+    },
+    [meshOverlayEditId, meshOverlayCurveAmount, meshOverlayCurveSegments]
+  );
+
+  /** Redondea o endereza (asas Bézier reversibles) la esquina del vértice. */
+  const overlayToggleCurveAt = useCallback(
+    (vertexIndex: number) => {
+      const editId = meshOverlayEditId;
+      if (editId == null) return;
+      setMeshSections((prev) =>
+        prev.map((s) => {
+          if (s.id !== editId) return s;
+          const poly = s.polygon;
+          const n = poly.length;
+          if (n < 3) return s;
+          const verts = poly.map((p) => ({ ...p }));
+          const v = verts[vertexIndex];
+          if (!v) return s;
+          if (v.hIn || v.hOut) {
+            delete v.hIn;
+            delete v.hOut;
+            return { ...s, polygon: verts };
+          }
+          const prevP = poly[(vertexIndex - 1 + n) % n];
+          const nextP = poly[(vertexIndex + 1) % n];
+          const d1x = v.x - prevP.x;
+          const d1y = v.y - prevP.y;
+          const d2x = nextP.x - v.x;
+          const d2y = nextP.y - v.y;
+          const k = meshOverlayCurveAmount;
+          v.hIn = { x: v.x - d1x * k, y: v.y - d1y * k };
+          v.hOut = { x: v.x + d2x * k, y: v.y + d2y * k };
+          return { ...s, polygon: verts };
+        })
+      );
+    },
+    [meshOverlayEditId, meshOverlayCurveAmount]
+  );
+
+  /** Pulsación sobre un tirador de asa Bézier: prepara su arrastre. */
+  const overlayHandlePointerDown = useCallback(
+    (id: number, vertex: number, side: 'in' | 'out') => {
+      const section = meshSections.find((s) => s.id === id);
+      const v = section?.polygon[vertex];
+      if (!section || !v) return;
+      const handle = side === 'in' ? v.hIn : v.hOut;
+      if (!handle) return;
+      overlayHandleRef.current = {
+        id,
+        vertex,
+        side,
+        anchor: { x: v.x, y: v.y },
+        other: side === 'in' ? (v.hOut ? { ...v.hOut } : null) : v.hIn ? { ...v.hIn } : null,
+        moved: false,
+        hadHandles: true,
+      };
+      setMeshOverlaySelection([vertex]);
+    },
+    [meshSections]
+  );
+
+  /** Crea las asas Bézier del vértice apuntando a sus vecinos (curva suave). */
+  const overlayCreateHandlesAt = useCallback(
+    (vertexIndex: number, amount: number) => {
+      const editId = meshOverlayEditId;
+      if (editId == null) return;
+      setMeshSections((prev) =>
+        prev.map((s) => {
+          if (s.id !== editId) return s;
+          const poly = s.polygon;
+          const n = poly.length;
+          if (n < 3) return s;
+          const verts = poly.map((p) => ({ ...p }));
+          const v = verts[vertexIndex];
+          if (!v) return s;
+          const prevP = poly[(vertexIndex - 1 + n) % n];
+          const nextP = poly[(vertexIndex + 1) % n];
+          v.hIn = {
+            x: v.x - (v.x - prevP.x) * amount,
+            y: v.y - (v.y - prevP.y) * amount,
+          };
+          v.hOut = {
+            x: v.x + (nextP.x - v.x) * amount,
+            y: v.y + (nextP.y - v.y) * amount,
+          };
+          return { ...s, polygon: verts };
+        })
+      );
+    },
+    [meshOverlayEditId]
+  );
+
+  /** Fija (o borra, con null) las asas indicadas del vértice. */
+  const overlayApplyHandles = useCallback(
+    (
+      id: number,
+      vertexIndex: number,
+      inPt: Handle2D | null,
+      outPt: Handle2D | null
+    ) => {
+      setMeshSections((prev) =>
+        prev.map((s) => {
+          if (s.id !== id) return s;
+          const verts = s.polygon.map((p) => ({ ...p }));
+          const v = verts[vertexIndex];
+          if (!v) return s;
+          if (inPt) v.hIn = { ...inPt };
+          else delete v.hIn;
+          if (outPt) v.hOut = { ...outPt };
+          else delete v.hOut;
+          return { ...s, polygon: verts };
+        })
+      );
+    },
+    []
+  );
+
+  /**
+   * Arrastra las asas del vértice según el puntero. `side` 'both' mueve las
+   * dos en oposición (salen a los dos lados); 'in'/'out' mueve una y, salvo
+   * que `symBreak` (tecla Alt) esté activo, mantiene la opuesta en espejo
+   * para que la curva siga siendo suave. Arrastrar hasta el vértice borra esa
+   * asa.
+   */
+  const overlayDragHandle = useCallback(
+    (clientX: number, clientY: number, symBreak: boolean) => {
+      const drag = overlayHandleRef.current;
+      if (!drag) return;
+      const { x, y } = overlayClientToView(clientX, clientY);
+      const px = Math.min(100, Math.max(0, overlaySnapView(x))) / 100;
+      const py = Math.min(100, Math.max(0, overlaySnapView(y))) / 100;
+      const dx = px - drag.anchor.x;
+      const dy = py - drag.anchor.y;
+      const len = Math.hypot(dx, dy);
+      const MIN = 0.02;
+      if (len < MIN) {
+        if (drag.side === 'in') {
+          overlayApplyHandles(drag.id, drag.vertex, null, drag.other);
+          drag.moved = true;
+        } else if (drag.side === 'out') {
+          overlayApplyHandles(drag.id, drag.vertex, drag.other, null);
+          drag.moved = true;
+        }
+        return;
+      }
+      drag.moved = true;
+      if (drag.side === 'both') {
+        overlayApplyHandles(
+          drag.id,
+          drag.vertex,
+          { x: drag.anchor.x - dx, y: drag.anchor.y - dy },
+          { x: px, y: py }
+        );
+        return;
+      }
+      const mirror =
+        !symBreak && drag.other
+          ? (() => {
+              const ox = drag.other.x - drag.anchor.x;
+              const oy = drag.other.y - drag.anchor.y;
+              const ol = Math.hypot(ox, oy) || len;
+              const k = ol / len;
+              return { x: drag.anchor.x - dx * k, y: drag.anchor.y - dy * k };
+            })()
+          : drag.other;
+      if (drag.side === 'in') {
+        overlayApplyHandles(drag.id, drag.vertex, { x: px, y: py }, mirror);
+      } else {
+        overlayApplyHandles(drag.id, drag.vertex, mirror, { x: px, y: py });
+      }
+    },
+    [overlayClientToView, overlaySnapView, overlayApplyHandles]
+  );
+
+  /** Pulsación sobre un tirador de vértice, según la herramienta activa. */
+  const overlayVertexPointerDown = useCallback(
+    (clientX: number, clientY: number, id: number, vertex: number) => {
+      if (meshOverlayTool === 'erase') {
+        overlayRemoveVertexAt(vertex);
+        return;
+      }
+      if (meshOverlayTool === 'select') {
+        const already = meshOverlaySelection.includes(vertex);
+        const sel = already ? meshOverlaySelection : [vertex];
+        if (!already) setMeshOverlaySelection(sel);
+        const section = meshSections.find((s) => s.id === id);
+        if (!section) return;
+        overlayGroupDragRef.current = {
+          id,
+          vertices: sel,
+          startView: overlayClientToView(clientX, clientY),
+          startPoly: section.polygon.map((p) => ({ ...p })),
+        };
+        return;
+      }
+      // Herramienta Editar: arrastre del vértice + pulsación larga para curvar.
+      // Con «Predibujar» activo, el clic sobre una esquina recta crea el arco
+      // directamente (la misma curva que se predibuja al pasar el ratón), igual
+      // que la herramienta Curva del editor 2D. Si ya está curvada, se arrastra.
+      if (meshOverlayCurvePreview) {
+        const section = meshSections.find((s) => s.id === id);
+        const v = section?.polygon[vertex];
+        if (section && v && !v.hIn && !v.hOut && section.polygon.length >= 3) {
+          overlayRoundCornerAt(vertex);
+          return;
+        }
+      }
+      const section = meshSections.find((s) => s.id === id);
+      const v = section?.polygon[vertex];
+      if (!section || !v || section.polygon.length < 3) return;
+      overlayDragRef.current = { id, vertex };
+      overlayHoldRef.current = {
+        timer: window.setTimeout(() => {
+          overlayDragRef.current = null;
+          overlayHoldRef.current = null;
+          const had = !!(v.hIn || v.hOut);
+          // Al agotarse la pulsación larga salen las asas: si la esquina era
+          // recta, se crean hacia sus vecinos (curva suave por defecto).
+          if (!had) overlayCreateHandlesAt(vertex, meshOverlayCurveAmount);
+          overlayHandleRef.current = {
+            id,
+            vertex,
+            side: 'both',
+            anchor: { x: v.x, y: v.y },
+            other: null,
+            moved: false,
+            hadHandles: had,
+          };
+        }, 380),
+        x: clientX,
+        y: clientY,
+      };
+    },
+    [
+      meshOverlayTool,
+      meshOverlaySelection,
+      meshSections,
+      overlayClientToView,
+      overlayRemoveVertexAt,
+      overlayCreateHandlesAt,
+      overlayRoundCornerAt,
+      meshOverlayCurvePreview,
+      meshOverlayCurveAmount,
+    ]
+  );
+
+  /** Arco previsto del redondeo del vértice bajo el cursor ("Predibujar"). */
+  const overlayHoverArc = useMemo(() => {
+    if (!meshOverlayCurvePreview || meshOverlayHoverVertex == null) return null;
+    const editId = meshOverlayEditId;
+    if (editId == null) return null;
+    const section = meshSections.find((s) => s.id === editId);
+    if (!section) return null;
+    const poly = section.polygon;
+    const n = poly.length;
+    if (n < 3) return null;
+    const i = meshOverlayHoverVertex;
+    if (i < 0 || i >= n) return null;
+    if (poly[i].hIn || poly[i].hOut) return null;
+    return cornerArcPoints(
+      poly,
+      i,
+      meshOverlayCurveAmount,
+      meshOverlayCurveSegments
+    );
+  }, [
+    meshOverlayCurvePreview,
+    meshOverlayHoverVertex,
+    meshOverlayEditId,
+    meshSections,
+    meshOverlayCurveAmount,
+    meshOverlayCurveSegments,
+  ]);
+
+  /** Mueve el vértice que se arrastra a la posición de pantalla indicada. */
+  const overlayMoveDraggedVertex = useCallback(
+    (clientX: number, clientY: number) => {
+      const drag = overlayDragRef.current;
+      if (!drag) return;
+      const { x, y } = overlayClientToView(clientX, clientY);
+      const nx = Math.min(100, Math.max(0, overlaySnapView(x))) / 100;
+      const ny = Math.min(100, Math.max(0, overlaySnapView(y))) / 100;
+      setMeshSections((prev) =>
+        prev.map((s) => {
+          if (s.id !== drag.id) return s;
+          const verts = s.polygon.map((p) => ({ ...p }));
+          const v = verts[drag.vertex];
+          if (!v) return s;
+          const dx = nx - v.x;
+          const dy = ny - v.y;
+          v.x = nx;
+          v.y = ny;
+          if (v.hIn) v.hIn = { x: v.hIn.x + dx, y: v.hIn.y + dy };
+          if (v.hOut) v.hOut = { x: v.hOut.x + dx, y: v.hOut.y + dy };
+          return { ...s, polygon: verts };
+        })
+      );
+    },
+    [overlayClientToView, overlaySnapView]
+  );
+
+  /** Inserta un vértice nuevo en la arista más cercana al punto pulsado. */
+  const overlayInsertVertexAt = useCallback(
+    (clientX: number, clientY: number) => {
+      const editId = meshOverlayEditId;
+      if (editId == null) return;
+      const { x, y } = overlayClientToView(clientX, clientY);
+      const px = Math.min(100, Math.max(0, overlaySnapView(x))) / 100;
+      const py = Math.min(100, Math.max(0, overlaySnapView(y))) / 100;
+      setMeshSections((prev) =>
+        prev.map((s) => {
+          if (s.id !== editId) return s;
+          const poly = s.polygon;
+          const n = poly.length;
+          if (n < 2) return s;
+          let bestI = 0;
+          let bestD = Infinity;
+          for (let i = 0; i < n; i++) {
+            const a = poly[i];
+            const b = poly[(i + 1) % n];
+            const dx = b.x - a.x;
+            const dy = b.y - a.y;
+            const lenSq = dx * dx + dy * dy || 1e-9;
+            const t = Math.max(
+              0,
+              Math.min(1, ((px - a.x) * dx + (py - a.y) * dy) / lenSq)
+            );
+            const qx = a.x + t * dx;
+            const qy = a.y + t * dy;
+            const d = Math.hypot(px - qx, py - qy);
+            if (d < bestD) {
+              bestD = d;
+              bestI = i;
+            }
+          }
+          const verts = poly.map((p) => ({ ...p }));
+          verts.splice(bestI + 1, 0, { x: px, y: py });
+          return { ...s, polygon: verts };
+        })
+      );
+    },
+    [meshOverlayEditId, overlayClientToView, overlaySnapView]
+  );
+
+  // Arrastre del vértice y desplazamiento del lienzo: los eventos viven en
+  // window para que el movimiento siga aunque el puntero salga del lienzo.
+  useEffect(() => {
+    const onMove = (e: PointerEvent) => {
+      const handleDrag = overlayHandleRef.current;
+      if (handleDrag) {
+        overlayDragHandle(e.clientX, e.clientY, e.altKey);
+        return;
+      }
+      const group = overlayGroupDragRef.current;
+      if (group) {
+        overlayMoveGroup(e.clientX, e.clientY);
+        return;
+      }
+      const marquee = overlayMarqueeRef.current;
+      if (marquee) {
+        const v = overlayClientToView(e.clientX, e.clientY);
+        marquee.x1 = v.x;
+        marquee.y1 = v.y;
+        setMeshOverlayMarquee({ ...marquee });
+        return;
+      }
+      if (overlayDragRef.current) {
+        const hold = overlayHoldRef.current;
+        if (hold) {
+          // Mientras no se supere el umbral sigue contando la pulsación larga.
+          if (Math.hypot(e.clientX - hold.x, e.clientY - hold.y) < 4) return;
+          window.clearTimeout(hold.timer);
+          overlayHoldRef.current = null;
+        }
+        overlayMoveDraggedVertex(e.clientX, e.clientY);
+        return;
+      }
+      const pan = overlayPanRef.current;
+      if (pan) {
+        const v = overlayClientToViewBox(e.clientX, e.clientY);
+        setMeshOverlayPan({
+          x: pan.panX + (v.x - pan.viewX),
+          y: pan.panY + (v.y - pan.viewY),
+        });
+      }
+    };
+    const onUp = () => {
+      const hold = overlayHoldRef.current;
+      if (hold) {
+        window.clearTimeout(hold.timer);
+        overlayHoldRef.current = null;
+      }
+      overlayDragRef.current = null;
+      const handleDrag = overlayHandleRef.current;
+      if (handleDrag) {
+        overlayHandleRef.current = null;
+        if (
+          handleDrag.side === 'both' &&
+          !handleDrag.moved &&
+          handleDrag.hadHandles
+        ) {
+          overlayToggleCurveAt(handleDrag.vertex);
+        }
+      }
+      overlayPanRef.current = null;
+      overlayGroupDragRef.current = null;
+      const marquee = overlayMarqueeRef.current;
+      if (marquee) {
+        overlayMarqueeRef.current = null;
+        overlayApplyMarquee(marquee.x0, marquee.y0, marquee.x1, marquee.y1);
+        setMeshOverlayMarquee(null);
+      }
+    };
+    window.addEventListener('pointermove', onMove);
+    window.addEventListener('pointerup', onUp);
+    window.addEventListener('pointercancel', onUp);
+    return () => {
+      window.removeEventListener('pointermove', onMove);
+      window.removeEventListener('pointerup', onUp);
+      window.removeEventListener('pointercancel', onUp);
+    };
+  }, [
+    overlayMoveDraggedVertex,
+    overlayMoveGroup,
+    overlayApplyMarquee,
+    overlayDragHandle,
+    overlayToggleCurveAt,
+    overlayClientToView,
+    overlayClientToViewBox,
+  ]);
+
+  /** Añade un vértice a la plantilla en edición (en su arista más larga). */
+  const overlayAddVertex = useCallback(() => {
+    if (meshOverlayEditId == null) return;
+    setMeshSections((prev) =>
+      prev.map((s) =>
+        s.id === meshOverlayEditId
+          ? { ...s, polygon: addPolygonVertex(s.polygon) }
+          : s
+      )
+    );
+  }, [meshOverlayEditId]);
+

+  const overlayRemoveVertex = useCallback(() => {
+    if (meshOverlayEditId == null) return;
+    setMeshSections((prev) =>
+      prev.map((s) =>
+        s.id === meshOverlayEditId
+          ? { ...s, polygon: removePolygonVertex(s.polygon) }
+          : s
+      )
+    );
+  }, [meshOverlayEditId]);
+

+        prev.map((s) =>
+          s.id === meshOverlayEditId

+    [meshOverlayEditId]

+        prev.map((s) =>
+          s.id === meshOverlayEditId

+    setMeshOverlayMarquee(null);

+    overlayHoldRef.current = null;
+    overlayGroupDragRef.current = null;
+  }, [meshOverlayEditId]);
+
+  /** Control deslizante de opacidad por plantilla (0..1). */
+  const renderOverlaySliders = (full: boolean) => (
+    <div
+      className={
+        full
+          ? 'shrink-0 grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-x-5 gap-y-1.5 max-h-[32vh] overflow-y-auto custom-scrollbar pr-1'
+          : 'space-y-1.5'
+      }
+    >
+      {meshSections.map((section, index) => {
+        const color =
+          SECTION_OVERLAY_COLORS[index % SECTION_OVERLAY_COLORS.length];
+        const op = overlayOpacityOf(section.id);

+            className={`flex items-center gap-2 rounded px-1 py-0.5 cursor-pointer transition-colors ${

+                : 'hover:bg-white/5'

+          >
+            <span
+              className="w-2.5 h-2.5 rounded-sm shrink-0 border border-white/20"
+              style={{ backgroundColor: color }}
+            />
+            {meshOverlayEditId === section.id && (
+              <Pencil className="w-2.5 h-2.5 text-green-400 shrink-0" />

+            <span className="text-[10px] text-foreground w-14 shrink-0">

+            <span className="font-mono text-[9px] text-green-400 w-12 shrink-0">

+              className="flex-1 flex items-center"
+              onClick={(e) => e.stopPropagation()}

+                max={1}
+                step={0.05}
+                value={[op]}
+                onValueChange={([v]) => setOverlayOpacity(section.id, v)}

+              />
+            </span>
+            <span className="font-mono text-[10px] text-muted-foreground w-8 text-right">
+              {Math.round(op * 100)}%
+            </span>
+          </div>
+        );
+      })}
+    </div>
+  );
+
+  /**
+   * Lienzo único con TODAS las plantillas superpuestas, en su color. También
+   * es el lienzo de EDICIÓN: la plantilla elegida (clic en su fila) se dibuja

+      <div
+        data-overlay-canvas
+        ref={setOverlayCanvas}

+          full ? 'flex-1 min-h-0' : heightClass
+        } rounded-lg overflow-hidden border border-white/10`}
+        style={{
+          background:
+            'radial-gradient(circle at 50% 40%, rgba(255,255,255,0.07), rgba(0,0,0,0.45))',
+        }}
+      >
+        <svg
+          viewBox="0 0 100 100"
+          preserveAspectRatio="xMidYMid meet"
+          className="absolute inset-0 w-full h-full select-none touch-none"

+            }

+              NO cambien al acercar o alejar. */}
+          <g
+            transform={`translate(${meshOverlayPan.x} ${meshOverlayPan.y}) translate(50 50) scale(${zoom}) translate(-50 -50)`}
+          >
+            {/* Rejilla magnética de fondo */}
+            {meshOverlayGrid && step > 0 && (
+              <g
+                stroke="rgba(255,255,255,0.10)"
+                strokeWidth={(full ? 0.15 : 0.25) / zoom}
+              >
+                {gridPositions.map((v) => (
+                  <line key={`gv${v}`} x1={v} y1={0} x2={v} y2={100} />
+                ))}
+                {gridPositions.map((v) => (
+                  <line key={`gh${v}`} x1={0} y1={v} x2={100} y2={v} />
+                ))}
+              </g>
+            )}
+
+            {/* Todas las plantillas; segmentos en marfil, relleno con su color */}
+            {meshSections.map((section, index) => {
+              const d = roundedPolygonPath(
+                section.polygon,
+                section.polygon.length >= 3
+              );
+              if (!d) return null;
+              const color =
+                SECTION_OVERLAY_COLORS[index % SECTION_OVERLAY_COLORS.length];
+              const isEdit = editSection?.id === section.id;
+              const baseOp = overlayOpacityOf(section.id);
+              const op = editSection && !isEdit ? baseOp * 0.35 : baseOp;
+              return (
+                <path
+                  key={section.id}
+                  d={d}
+                  fill={color}
+                  fillOpacity={isEdit ? op * 0.25 : op * 0.35}
+                  stroke={OVERLAY_SEGMENT_IVORY}
+                  strokeOpacity={isEdit ? 1 : op}
+                  strokeWidth={
+                    (isEdit ? (full ? 0.2 : 0.3) : full ? 0.11 : 0.17) / zoom
+                  }
+                  strokeLinejoin="round"
+                  strokeLinecap="round"
+                />
+              );
+            })}
+
+            {/* Tiradores de vértice de la plantilla en edición */}
+            {editSection &&
+              editSection.polygon.map((p, i) => {
+                const ci = meshSections.findIndex(
+                  (s) => s.id === editSection.id
+                );
+                const color =
+                  SECTION_OVERLAY_COLORS[
+                    (ci < 0 ? 0 : ci) % SECTION_OVERLAY_COLORS.length
+                  ];
+                const isSelected = meshOverlaySelection.includes(i);
+                return (
+                  <g key={`v${i}`}>
+                    <circle
+                      cx={p.x * 100}
+                      cy={p.y * 100}
+                      r={(full ? 2.2 : 3) / zoom}
+                      fill="transparent"
+                      style={{
+                        cursor:
+                          meshOverlayTool === 'erase' ? 'crosshair' : 'default',
+                      }}
+                      onPointerDown={(e) => {
+                        e.stopPropagation();
+                        overlayVertexPointerDown(
+                          e.clientX,
+                          e.clientY,
+                          editSection.id,
+                          i
+                        );
+                      }}
+                      onPointerEnter={() => setMeshOverlayHoverVertex(i)}
+                      onPointerLeave={() =>
+                        setMeshOverlayHoverVertex((h) => (h === i ? null : h))
+                      }
+                    />
+                    {p.hIn && (
+                      <g>
+                        <line
+                          x1={p.x * 100}
+                          y1={p.y * 100}
+                          x2={p.hIn.x * 100}
+                          y2={p.hIn.y * 100}
+                          stroke="#22c55e"
+                          strokeOpacity={0.7}
+                          strokeWidth={(full ? 0.12 : 0.18) / zoom}
+                          strokeDasharray={`${0.5 / zoom} ${0.5 / zoom}`}
+                          pointerEvents="none"
+                        />
+                        <circle
+                          cx={p.hIn.x * 100}
+                          cy={p.hIn.y * 100}
+                          r={(full ? 1.9 : 2.6) / zoom}
+                          fill="transparent"
+                          style={{ cursor: 'default' }}
+                          onPointerDown={(e) => {
+                            e.stopPropagation();
+                            overlayHandlePointerDown(editSection.id, i, 'in');
+                          }}
+                        />
+                        <circle
+                          cx={p.hIn.x * 100}
+                          cy={p.hIn.y * 100}
+                          r={(full ? 0.38 : 0.55) / zoom}
+                          fill="#22c55e"
+                          stroke="#bbf7d0"
+                          strokeWidth={(full ? 0.14 : 0.2) / zoom}
+                          pointerEvents="none"
+                        />
+                      </g>
+                    )}
+                    {p.hOut && (
+                      <g>
+                        <line
+                          x1={p.x * 100}
+                          y1={p.y * 100}
+                          x2={p.hOut.x * 100}
+                          y2={p.hOut.y * 100}
+                          stroke="#22c55e"
+                          strokeOpacity={0.7}
+                          strokeWidth={(full ? 0.12 : 0.18) / zoom}
+                          strokeDasharray={`${0.5 / zoom} ${0.5 / zoom}`}
+                          pointerEvents="none"
+                        />
+                        <circle
+                          cx={p.hOut.x * 100}
+                          cy={p.hOut.y * 100}
+                          r={(full ? 1.9 : 2.6) / zoom}
+                          fill="transparent"
+                          style={{ cursor: 'default' }}
+                          onPointerDown={(e) => {
+                            e.stopPropagation();
+                            overlayHandlePointerDown(editSection.id, i, 'out');
+                          }}
+                        />
+                        <circle
+                          cx={p.hOut.x * 100}
+                          cy={p.hOut.y * 100}
+                          r={(full ? 0.38 : 0.55) / zoom}
+                          fill="#22c55e"
+                          stroke="#bbf7d0"
+                          strokeWidth={(full ? 0.14 : 0.2) / zoom}
+                          pointerEvents="none"
+                        />
+                      </g>
+                    )}
+                    {meshOverlayCurvePreview &&
+                      meshOverlayHoverVertex === i &&
+                      overlayHoverArc && (
+                        <polyline
+                          points={overlayHoverArc
+                            .map((q) => `${q.x * 100},${q.y * 100}`)

+                          fill="none"
+                          stroke="#22c55e"
+                          strokeWidth={(full ? 0.3 : 0.45) / zoom}
+                          strokeDasharray={`${0.8 / zoom} ${0.8 / zoom}`}
+                          pointerEvents="none"

+                      cx={p.x * 100}
+                      cy={p.y * 100}
+                      r={

+                        zoom
+                      }
+                      fill={isSelected ? '#22c55e' : '#ffffff'}
+                      stroke={isSelected ? '#bbf7d0' : color}
+                      strokeWidth={(full ? 0.35 : 0.5) / zoom}
+                      pointerEvents="none"
+                    />
+                  </g>
+                );
+              })}
+
+            {/* Marquesina de selección (herramienta Selección) */}
+            {meshOverlayMarquee && (
+              <rect
+                x={Math.min(meshOverlayMarquee.x0, meshOverlayMarquee.x1)}
+                y={Math.min(meshOverlayMarquee.y0, meshOverlayMarquee.y1)}
+                width={Math.abs(meshOverlayMarquee.x1 - meshOverlayMarquee.x0)}
+                height={Math.abs(meshOverlayMarquee.y1 - meshOverlayMarquee.y0)}
+                fill="rgba(34,197,94,0.12)"
+                stroke="#22c55e"
+                strokeWidth={0.3 / zoom}
+                strokeDasharray={`${0.7 / zoom} ${0.5 / zoom}`}
+                pointerEvents="none"
+              />
+            )}
+          </g>
+        </svg>
+      </div>
+    );
+  };
+
+  /** Barra de herramientas de edición de la vista superpuesta (Mallas). */
+  const renderOverlayEditTools = () => {
+    const editIndex =
+      meshOverlayEditId != null
+        ? meshSections.findIndex((s) => s.id === meshOverlayEditId)
+        : -1;
+    const hasSel = editIndex >= 0;
+    return (
+      <div className="shrink-0 flex flex-wrap items-center gap-1.5">
+        <span className="text-[10px] text-muted-foreground/70 flex items-center gap-1 min-w-0">
+          <MousePointer2 className="w-3 h-3 shrink-0" />
+          <span className="truncate">
+            {hasSel
+              ? t('editor3D.overlayEditing', { n: editIndex + 1 })
+              : t('editor3D.overlayPickHint')}
+          </span>
+        </span>
+        <div className="flex-1" />
+        <button
+          type="button"
+          onClick={undo}
+          data-testid="overlay-undo-btn"
+          disabled={historyIndex <= 0}
+          title={t('editor3D.undoTitle')}
+          className={overlayToolChip(false, historyIndex <= 0)}

+          <Undo2 className="w-3 h-3" />
+        </button>

+          type="button"
+          onClick={redo}
